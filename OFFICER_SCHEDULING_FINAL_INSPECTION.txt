function localDateTime(value: Date) {
  const pad = (n: number) => String(n).padStart(2, "0");
  return `${value.getFullYear()}-${pad(value.getMonth() + 1)}-${pad(value.getDate())}T${pad(value.getHours())}:${pad(value.getMinutes())}`;
}

export function makeAccess(roles: AppRole[], permissions: string[]): AccessState {
  const isSuperAdmin = roles.includes("super_admin");
  return {
    roles,
    permissions: new Set(isSuperAdmin ? ALL_PERMISSIONS : permissions),

  };
}

export function allows(access: AccessState | undefined | null, permission: string): boolean {
  if (!access) return false;
  return access.isSuperAdmin || access.permissions.has(permission);
}

export function primaryRole(roles: AppRole[]): AppRole | null {
  for (const role of ALL_ROLES) if (roles.includes(role)) return role;
  return null;
}

import { useQuery, useQueryClient } from "@tanstack/react-query";
import { supabase } from "@/integrations/supabase/client";
import { makeAccess, type AccessState, type AppRole } from "@/lib/rbac";

export interface StaffProfile {
  id: string;
  email: string;
  full_name: string;
  phone: string | null;
  department: string | null;
  job_title: string | null;
  photo_url: string | null;
  status: "active" | "disabled" | "pending";
  must_change_password: boolean;
  last_login_at: string | null;
  created_at: string;
}

export interface SessionState {
  userId: string;
  email: string | null;
  profile: StaffProfile | null;
  roles: AppRole[];
  access: AccessState;
}

export async function fetchSession(): Promise<SessionState | null> {
  const { data: userData, error: authError } = await supabase.auth.getUser();
  if (authError) throw authError;
  const user = userData.user;
  if (!user) return null;

  const [profileResult, rolesResult, permissionsResult] = await Promise.all([
    supabase.from("profiles").select("*").eq("id", user.id).maybeSingle(),
    supabase.from("user_roles").select("role").eq("user_id", user.id),
    supabase.rpc("my_permissions"),
  ]);
  const requestError = profileResult.error ?? rolesResult.error ?? permissionsResult.error;
  if (requestError) throw requestError;

  const roleList = (rolesResult.data ?? []).map((role) => role.role as AppRole);
  const permissionList = ((permissionsResult.data ?? []) as { permission: string }[]).map(
    (permission) => permission.permission,
  );
  return {
    userId: user.id,
    email: user.email ?? null,
    profile: (profileResult.data as StaffProfile) ?? null,
    roles: roleList,
    access: makeAccess(roleList, permissionList),
  };
}

export function useSession() {
  const query = useQuery<SessionState | null>({
    queryKey: ["session"],
    queryFn: fetchSession,
    staleTime: 30_000,
  });

  return query;
}

export function useAccess() {
  const { data } = useSession();
  return data?.access;
}

export function useRefreshSession() {
  const queryClient = useQueryClient();
  return () => queryClient.invalidateQueries({ queryKey: ["session"] });
}

export async function logAudit(input: {
  actorId: string;
  actorEmail?: string | null;
  action: string;
  entityType?: string;
  entityId?: string | null;
  newValue?: unknown;
}) {
  await supabase.from("audit_logs").insert({
    actor_id: input.actorId,
    actor_email: input.actorEmail ?? null,
    action: input.action,
    entity_type: input.entityType ?? null,
    entity_id: input.entityId ?? null,
    new_value: (input.newValue ?? null) as never,
  });
}

import { createClient } from '@supabase/supabase-js';
import type { Database } from './types';

function isNewSupabaseApiKey(value: string): boolean {
  return value.startsWith('sb_publishable_') || value.startsWith('sb_secret_');
}

function createSupabaseFetch(supabaseKey: string): typeof fetch {
  return (input, init) => {
    const headers = new Headers(
      typeof Request !== 'undefined' && input instanceof Request ? input.headers : undefined,
    );

    if (init?.headers) {
      new Headers(init.headers).forEach((value, key) => headers.set(key, value));
    }

    // New Supabase API keys are opaque strings, not bearer JWTs.
    if (isNewSupabaseApiKey(supabaseKey) && headers.get('Authorization') === `Bearer ${supabaseKey}`) {
      headers.delete('Authorization');
    }

    headers.set('apikey', supabaseKey);
    return fetch(input, { ...init, headers });
  };
}


function createSupabaseClient() {
  // Use import.meta.env for client-side (Vite build-time replacement)
  // Fall back to process.env for SSR (server-side rendering)
  const SUPABASE_URL = import.meta.env['VITE_SUPABASE_URL'] || process.env['SUPABASE_URL'];
  const SUPABASE_PUBLISHABLE_KEY = import.meta.env['VITE_SUPABASE_PUBLISHABLE_KEY'] || process.env['SUPABASE_PUBLISHABLE_KEY'];

  if (!SUPABASE_URL || !SUPABASE_PUBLISHABLE_KEY) {
    const missing = [
      ...(!SUPABASE_URL ? ['SUPABASE_URL'] : []),
      ...(!SUPABASE_PUBLISHABLE_KEY ? ['SUPABASE_PUBLISHABLE_KEY'] : []),
    ];
    const message = `Missing Supabase environment variable(s): ${missing.join(', ')}. Set them in .env (local) or Vercel project settings.`;
    console.error(`[Supabase] ${message}`);
    throw new Error(message);
  }

  return createClient<Database>(SUPABASE_URL, SUPABASE_PUBLISHABLE_KEY, {
    global: {
      fetch: createSupabaseFetch(SUPABASE_PUBLISHABLE_KEY),
    },
    auth: {
      storage: typeof window !== 'undefined' ? window.localStorage : undefined,
      persistSession: true,
      autoRefreshToken: true,
    }
  });
}

let _supabase: ReturnType<typeof createSupabaseClient> | undefined;

// Import the supabase client like this:
// import { supabase } from "@/integrations/supabase/client";
export const supabase = new Proxy({} as ReturnType<typeof createSupabaseClient>, {
  get(_, prop, receiver) {
    if (!_supabase) _supabase = createSupabaseClient();
    return Reflect.get(_supabase, prop, receiver);
  },
});
