import { useEffect, useRef } from "react";
import { useQuery } from "@tanstack/react-query";
import { toast } from "sonner";
import { supabase } from "@/integrations/supabase/client";
import { useSession } from "@/hooks/useSession";
import { allows } from "@/lib/rbac";
import { enableNotificationSound, isNotificationSoundEnabled, playNotificationSound } from "@/lib/notification-sound";

function localDateKey(date: Date): string {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");
  return `${year}-${month}-${day}`;
}

export function DailyOperationsBriefing() {
  const announcedDays = useRef(new Set<string>());
  const { data: session } = useSession();
  const day = localDateKey(new Date());
  const canViewObjectives = allows(session?.access, "objectives.view");
  const canViewCalendar = allows(session?.access, "clients.view");
  const canViewTasks = allows(session?.access, "tasks.view");

  const briefing = useQuery({
    queryKey: ["daily-operations-briefing", session?.userId, day],
    enabled: Boolean(session?.userId && (canViewObjectives || canViewCalendar || canViewTasks)),
    queryFn: async () => {
      const start = new Date();
      start.setHours(0, 0, 0, 0);
      const end = new Date(start);
      end.setDate(end.getDate() + 1);

      const [objectivesResult, eventsResult, tasksResult] = await Promise.all([
        canViewObjectives
          ? supabase
              .from("objectives")
              .select("id,title,target_date,progress,status")
              .is("archived_at", null)
              .in("status", ["planned", "in_progress"])
              .order("target_date", { ascending: true })
              .limit(10)
          : Promise.resolve({ data: [], error: null }),
        canViewCalendar
          ? supabase
              .from("calendar_events")
              .select("id,title,starts_at")
              .gte("starts_at", start.toISOString())
              .lt("starts_at", end.toISOString())
              .order("starts_at", { ascending: true })
              .limit(10)
          : Promise.resolve({ data: [], error: null }),
        canViewTasks
          ? supabase
              .from("tasks")
              .select("id,title,due_date,status")
              .eq("assigned_to", session!.userId)
              .eq("due_date", day)
              .is("archived_at", null)
              .order("due_date", { ascending: true })
              .limit(10)
          : Promise.resolve({ data: [], error: null }),
      ]);

      const error = objectivesResult.error ?? eventsResult.error ?? tasksResult.error;
      if (error) throw error;

      return {
        objectives: objectivesResult.data ?? [],
        events: eventsResult.data ?? [],
        tasks: (tasksResult.data ?? []).filter(
          (task) => !["completed", "complete", "done"].includes(task.status.toLowerCase()),
        ),
      };
    },
    refetchInterval: 60_000,
    refetchOnWindowFocus: true,
    retry: 1,
  });

  useEffect(() => {
    if (!session?.userId || !briefing.data) return;
    const storageKey = `kalexi-daily-briefing-${session.userId}-${day}`;
    if (announcedDays.current.has(storageKey)) return;
    try {
      if (window.localStorage.getItem(storageKey)) {
        announcedDays.current.add(storageKey);
        return;
      }
    } catch (error) {
      console.warn("[daily briefing] Could not read the reminder preference.", error);
    }

    const { objectives, events, tasks } = briefing.data;
    const lines = [
      `${objectives.length} active company objective${objectives.length === 1 ? "" : "s"}`,
      `${events.length} calendar event${events.length === 1 ? "" : "s"} today`,
      `${tasks.length} task${tasks.length === 1 ? "" : "s"} due today`,
    ];
    const objectiveTitles = objectives.slice(0, 3).map((objective) => objective.title);
    const eventTitles = events.slice(0, 3).map((event) => event.title);
    const taskTitles = tasks.slice(0, 3).map((task) => task.title);
    const details = [
      ...objectiveTitles.map((title) => `Objective: ${title}`),
      ...eventTitles.map((title) => `Event: ${title}`),
      ...taskTitles.map((title) => `Due: ${title}`),
    ];

    announcedDays.current.add(storageKey);
    try {
      window.localStorage.setItem(storageKey, "shown");
    } catch (error) {
      console.warn("[daily briefing] Could not store the reminder date.", error);
    }
    if (isNotificationSoundEnabled()) playNotificationSound();
    toast.info("Daily company briefing", {
      description: `${lines.join(" Â· ")}${details.length ? `\n${details.join(" Â· ")}` : ""}`,
      duration: 12_000,
      action: isNotificationSoundEnabled()
        ? undefined
        : {
            label: "Enable sound",
            onClick: enableNotificationSound,
          },
    });
  }, [briefing.data, day, session?.userId]);

  useEffect(() => {
    if (briefing.isError && briefing.error) {
      console.error("[daily briefing] Could not load company reminders.", briefing.error);
    }
  }, [briefing.error, briefing.isError]);

  return null;
}

async function openCampaign(t: string) {
  const db = await admin();
  const { data, error } = await db
    .from("agent_campaigns")
    .select("id, title, active, expires_at")
    .eq("token", t)
    .maybeSingle();
  if (error) throw new Error("Could not verify this registration link.");
  let campaign = data;
  if (!campaign && /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(t)) {
    const legacy = await db
      .from("agent_campaigns")
      .select("id, title, active, expires_at")
      .eq("id", t)
      .maybeSingle();
    if (legacy.error) throw new Error("Could not verify this registration link.");
    campaign = legacy.data;
  }
  if (!campaign) return null;
  const expired = !!campaign.expires_at && new Date(campaign.expires_at) < new Date();
  return { ...campaign, open: campaign.active && !expired };
}

async function assertPerm(supabase: SupabaseClient<Database>, userId: string, perm: string) {
  const { data, error } = await supabase.rpc("has_permission", { _user_id: userId, _perm: perm });
  if (error) throw new Error("Could not verify Digital Agent Programme permissions.");
  if (!data) throw new Error("Forbidden: you do not have Digital Agent Programme authority.");
}

async function assertAnyPerm(
  supabase: SupabaseClient<Database>,
  userId: string,
  permissions: string[],
) {
  const checks = await Promise.all(
    permissions.map((permission) =>
      supabase.rpc("has_permission", { _user_id: userId, _perm: permission }),
    ),
  );
  if (checks.some(({ error }) => error))
    throw new Error("Could not verify Digital Agent Programme permissions.");
  if (!checks.some(({ data }) => data))
    throw new Error("Forbidden: you do not have Digital Agent Programme authority.");
}

  roles: AppRole[];
  permissions: Set<string>;

export function makeAccess(roles: AppRole[], permissions: string[]): AccessState {
  const isSuperAdmin = roles.includes("super_admin");
  return {
    roles,
    permissions: new Set(isSuperAdmin ? ALL_PERMISSIONS : permissions),
    isSuperAdmin,
  };
}

export function allows(access: AccessState | undefined | null, permission: string): boolean {
  if (!access) return false;
  return access.isSuperAdmin || access.permissions.has(permission);
}

export function primaryRole(roles: AppRole[]): AppRole | null {
  for (const role of ALL_ROLES) if (roles.includes(role)) return role;
  return null;
}

function localDateTime(value: string) {
  if (!value) return "";
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return value;
  return date.toLocaleString("en-UG", {
    timeZone: "Africa/Kampala",
    dateStyle: "medium",
    timeStyle: "short",
  });
}

function localDateTime(value: Date) {
  const pad = (n: number) => String(n).padStart(2, "0");
  return `${value.getFullYear()}-${pad(value.getMonth() + 1)}-${pad(value.getDate())}T${pad(value.getHours())}:${pad(value.getMinutes())}`;
}

CREATE OR REPLACE FUNCTION public.prevent_officer_schedule_overlap()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  conflict_found boolean := false;
BEGIN
  -- Serialize concurrent schedule writes for the same officer.
  PERFORM pg_advisory_xact_lock(hashtextextended(NEW.officer_id::text, 0));

  IF TG_TABLE_NAME = 'officer_appointments' THEN
    IF NEW.status IN ('pending','confirmed') THEN
      SELECT EXISTS (
        SELECT 1 FROM public.officer_appointments a
        WHERE a.officer_id = NEW.officer_id
          AND a.id <> NEW.id
          AND a.status IN ('pending','confirmed')
          AND tstzrange(a.starts_at, a.ends_at, '[)') && tstzrange(NEW.starts_at, NEW.ends_at, '[)')
      ) OR EXISTS (
        SELECT 1 FROM public.officer_programmes p
        WHERE p.officer_id = NEW.officer_id
          AND p.status = 'scheduled'
          AND tstzrange(p.starts_at, p.ends_at, '[)') && tstzrange(NEW.starts_at, NEW.ends_at, '[)')
      ) INTO conflict_found;
    END IF;
  ELSE
    IF NEW.status = 'scheduled' THEN
      SELECT EXISTS (
        SELECT 1 FROM public.officer_programmes p
        WHERE p.officer_id = NEW.officer_id
          AND p.id <> NEW.id
          AND p.status = 'scheduled'
          AND tstzrange(p.starts_at, p.ends_at, '[)') && tstzrange(NEW.starts_at, NEW.ends_at, '[)')
      ) OR EXISTS (
        SELECT 1 FROM public.officer_appointments a
        WHERE a.officer_id = NEW.officer_id
          AND a.status IN ('pending','confirmed')
          AND tstzrange(a.starts_at, a.ends_at, '[)') && tstzrange(NEW.starts_at, NEW.ends_at, '[)')
      ) INTO conflict_found;
    END IF;
  END IF;

  IF conflict_found THEN
    RAISE EXCEPTION 'OFFICER_TIME_CONFLICT: This officer already has a booking or programme during the selected time.'
      USING ERRCODE = '23P01';
  END IF;
  NEW.updated_at := now();
  RETURN NEW;
END;
$$;

  officer_id = auth.uid() OR
  public.has_role(auth.uid(), 'receptionist'::public.app_role) OR
  public.has_role(auth.uid(), 'management'::public.app_role) OR
  public.has_role(auth.uid(), 'super_admin'::public.app_role)
);

DROP POLICY IF EXISTS officer_appointments_reception_insert ON public.officer_appointments;
CREATE POLICY officer_appointments_reception_insert ON public.officer_appointments
FOR INSERT TO authenticated
WITH CHECK (
  created_by = auth.uid() AND (
    public.has_role(auth.uid(), 'receptionist'::public.app_role) OR
    public.has_role(auth.uid(), 'management'::public.app_role) OR
    public.has_role(auth.uid(), 'super_admin'::public.app_role)
  )
);

DROP POLICY IF EXISTS officer_appointments_reception_update ON public.officer_appointments;
CREATE POLICY officer_appointments_reception_update ON public.officer_appointments
FOR UPDATE TO authenticated
USING (
  public.has_role(auth.uid(), 'receptionist'::public.app_role) OR
  public.has_role(auth.uid(), 'management'::public.app_role) OR
  public.has_role(auth.uid(), 'super_admin'::public.app_role)
)
WITH CHECK (
  public.has_role(auth.uid(), 'receptionist'::public.app_role) OR
  public.has_role(auth.uid(), 'management'::public.app_role) OR
  public.has_role(auth.uid(), 'super_admin'::public.app_role)
);

GRANT SELECT, INSERT, UPDATE ON public.officer_programmes TO authenticated;
GRANT SELECT, INSERT, UPDATE ON public.officer_appointments TO authenticated;
GRANT ALL ON public.officer_programmes, public.officer_appointments TO service_role;
